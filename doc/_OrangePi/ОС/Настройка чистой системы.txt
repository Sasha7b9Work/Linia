 1. Установить XFCE
    sudo apt install xfce4 xfce4-goodies

 2. Настроить автовход при загрузке:
     # Проверить наличие различных менеджеров входа
     ls /usr/sbin/ | grep -E "lightdm|gdm3|sddm|lxdm|xdm"
     и в зависимости от установленного менеджера действовать так, чтобы был автовход с сессией xfce
     
     lightdm:
     sudo nano /etc/lightdm/lightdm.conf.d/50-autologin.conf
        [SeatDefaults]
        autologin-user=your_username
        autologin-session=xfce
        autologin-user-timeout=0
 
 3. Установить Chickago95:
    cd ~
    git clone https://github.com/grassmunk/chicago95
    cd chicago95
    ./installer.py
    !!! ВНИМАНИЕ !!! Запускать инсталлятор нужно с рабочего стола
    
    После установки можно зайти в .bashrc и удалить последние строки, где настраивается вид приглашения как в Windows
 
 4. Отформатировать и примонтировать nvme диск в /mnt/nvme

 5. Склонировать Linia2 в /mnt/nvme

 6. Сканирование диапазона портов на наличие устройств
    fping -ag 192.168.1.0/24 2>/dev/null)

 7. Скопировать в /home/mnipi/Desktop скрипт из Автозагрузка/Desktop
 
 8. Включить spi и uart :
    armbian-config
    
    boot/armbianEnv.txt:
        verbosity=1
        bootlogo=true
        console=both
        extraargs=cma=256M
        overlay_prefix=rockchip-rk3588
        fdtfile=rockchip/rk3588-orangepi-5-plus.dtb
        rootdev=UUID=c80228b6-df38-4ce2-bb64-8ce362822ffe
        rootfstype=ext4
        overlays=rk3588-spi0-m2-cs0-spidev rk3588-uart6-m1 rk3588-uart6-m1-full
        usbstoragequirks=0x2537:0x1066:u,0x2537:0x1068:u

    
    Проверка UART6:
        echo "Hello UART6" | sudo tee /dev/ttyS6
        на контактах 8, 10 должен быть сигнал
    
 9. Дать права на доступ к аппаратным ресурсам пользователю mnipi:
 
        GPIO:
    sudo groupadd gpio
    sudo usermod -aG gpio mnipi
    sudo nano /etc/udev/rules.d/99-gpio.rules
        SUBSYSTEM=="gpio", GROUP="gpio", MODE="0660"
        SUBSYSTEM=="gpio", KERNEL=="gpiochip*", GROUP="gpio", MODE="0660"
    sudo udevadm control --reload-rules
    sudo udevadm trigger
        Проверяем права:
    ls -l /dev/gpiochip*
        crw-rw---- 1 root gpio 254, 0 ... /dev/gpiochip0
        crw-rw---- 1 root gpio 254, 1 ... /dev/gpiochip1
        
        SPI:
    sudo groupadd spi
    sudo usermod -aG spi mnipi
    sudo nano /etc/udev/rules.d/99-spi.rules
        SUBSYSTEM=="spidev", GROUP="spi", MODE="0660"
    sudo udevadm control --reload-rules
    sudo udevadm trigger -s spidev
    ls -l /dev/spidev*
        crw-rw---- 1 root spi 153, 0 ... /dev/spidev0.0
    
    sudo reboot
    
10. Автозапуск приложение после входа в систему:
        mkdir -p ~/.config/autostart
        cp ~/Desktop/Linia.desktop ~/.config/autostart/
